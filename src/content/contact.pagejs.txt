function handleSubmit(e: Event) {
  e.preventDefault();
  const val = (id: string) => {
    const el = document.getElementById(id) as HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement | null;
    return el ? el.value : "";
  };
  const fname = val("fname").trim();
  const lname = val("lname").trim();
  const email = val("email").trim();
  const subject = val("subject").trim();
  const message = val("message").trim();
  const msg = document.getElementById("formMsg");
  const show = (t: string) => {
    if (msg) {
      msg.style.display = "block";
      msg.innerHTML = t;
    }
  };
  if (!fname || !lname) { show("Please enter your first and last name."); return; }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { show("Please enter a valid email address."); return; }
  if (!subject) { show("Please select a topic."); return; }
  if (!message) { show("Please write your message."); return; }
  const body = "Name: " + fname + " " + lname + "\nEmail: " + email + "\n\n" + message;
  window.location.href =
    "mailto:contact@crscalculator.site?subject=" + encodeURIComponent(subject) +
    "&body=" + encodeURIComponent(body);
  show('Your email app should open. If it doesn\u2019t, write to <a href="mailto:contact@crscalculator.site">contact@crscalculator.site</a>.');
}
