    /* NAV TOGGLE */
    function toggleNav(btn) {
      const nav = document.getElementById('navLinks');
      const open = nav?.classList.toggle('open');
      btn.setAttribute('aria-expanded', String(open));
    }

    /* SECTION COLLAPSE */
    function toggleSection(btn) {
      const expanded = btn.getAttribute('aria-expanded') === 'true';
      const bodyId = btn.getAttribute('aria-controls');
      const body = bodyId ? document.getElementById(bodyId) : btn.nextElementSibling;
      if (!body) return;
      const newExpanded = !expanded;
      btn.setAttribute('aria-expanded', String(newExpanded));
      body.style.display = newExpanded ? '' : 'none';
      const arrow = btn.querySelector('.section-arrow');
      if (arrow) arrow.style.transform = newExpanded ? '' : 'rotate(-90deg)';
    }

    /* SPOUSE TOGGLE */
    function toggleSpouse(cb) {
      const sf = document.getElementById('spouseFields');
      sf.style.display = cb.checked ? 'block' : 'none';
      sf.setAttribute('aria-hidden', String(!cb.checked));
    }

    /* LANG HINTS */

      const hints = {

    /* IELTS → CLB (per skill) */
    function ieltsToClbBySkill(score, skill) {
      const maps = {
        listening: [[8.5,12],[8.0,10],[7.5,9],[6.0,8],[5.5,7],[5.0,6],[4.5,5],[4.0,4]],
        reading:   [[8.5,12],[8.0,10],[7.0,9],[6.5,8],[6.0,7],[5.0,6],[4.0,5],[3.5,4]],
        writing:   [[7.5,12],[7.0,10],[6.5,9],[6.0,8],[5.5,7],[5.0,6],[4.0,5],[3.5,4]],
        speaking:  [[7.5,12],[7.0,10],[6.5,9],[6.0,8],[5.5,7],[5.0,6],[4.0,5],[3.5,4]]
      };
      const map = maps[skill] || maps.reading;
      for (const [threshold, clb] of map) {
        if (score >= threshold) return clb;
      }
      return 3;
    }

    /* CLB → CRS Language Points */
    function clbToLangPts(clb, isFirst, hasSpouse) {
      const s1 = {3:0,4:6,5:6,6:9,7:17,8:23,9:31,10:34,11:34,12:34};
      const c1 = {3:0,4:6,5:6,6:8,7:16,8:22,9:29,10:32,11:32,12:32};
      const s2 = {3:0,4:0,5:1,6:1,7:3,8:3,9:6,10:6,11:6,12:6};
      const t = hasSpouse ? c1 : (isFirst ? s1 : s2);
      return t[Math.min(clb, 12)] || 0;
    }

    /* MAIN CALCULATE */
    function calculateCRS() {
      const hasSpouse = document.getElementById('spouseToggle').checked;
      const ageVal = document.getElementById('age').value;

      /* ── Section A ── */
      const agePts   = parseInt(ageVal) || 0;
      const eduPts   = parseInt(document.getElementById('education').value) || 0;
      const canExpPts = parseInt(document.getElementById('canadaExp').value) || 0;

      /* Language */
      const testType = document.getElementById('langTest').value;
      const lsV = parseFloat(document.getElementById('ls').value) || 0;
      const rsV = parseFloat(document.getElementById('rs').value) || 0;
      const wsV = parseFloat(document.getElementById('ws').value) || 0;
      const ssV = parseFloat(document.getElementById('ss').value) || 0;

      let langPts = 0;
      let avgClb = 0;
      if (testType === 'ielts' || testType === 'celpip') {
        const clbL = ieltsToClbBySkill(lsV, 'listening');
        const clbR = ieltsToClbBySkill(rsV, 'reading');
        const clbW = ieltsToClbBySkill(wsV, 'writing');
        const clbS = ieltsToClbBySkill(ssV, 'speaking');
        avgClb = Math.round((clbL + clbR + clbW + clbS) / 4);
        langPts = clbToLangPts(clbL,'first',hasSpouse)
                + clbToLangPts(clbR,'first',hasSpouse)
                + clbToLangPts(clbW,'first',hasSpouse)
                + clbToLangPts(clbS,'first',hasSpouse);
      } else {
        // TEF/TCF simplified
        const avg = (lsV + rsV + wsV + ssV) / 4;
        avgClb = Math.min(12, Math.max(4, Math.round(avg / 100 * 5 + 5)));
        langPts = clbToLangPts(avgClb,'first',hasSpouse) * 4;
      }

      const sectionA = agePts + eduPts + canExpPts + langPts;

      /* ── Section B (Spouse) ── */
      let sectionB = 0;
      if (hasSpouse) {
        sectionB += parseInt(document.getElementById('spouseEdu').value) || 0;
        sectionB += parseInt(document.getElementById('spouseCanExp').value) || 0;
        sectionB = Math.min(sectionB, 40);
      }

      /* ── Section C (Skill Transferability) ── */
      const foreignExpPts = parseInt(document.getElementById('foreignExp').value) || 0;
      let sectionC = 0;
      // Education + language
      if (avgClb >= 9 && eduPts >= 112) sectionC += 25;
      else if (avgClb >= 7 && eduPts >= 84) sectionC += 13;
      // Canadian work exp + language
      if (canExpPts > 0 && avgClb >= 9) sectionC += 25;
      else if (canExpPts > 0 && avgClb >= 7) sectionC += 13;
      // Foreign work + Canadian work
      if (foreignExpPts > 0 && canExpPts > 0) sectionC += foreignExpPts === 25 ? 25 : 13;
      sectionC = Math.min(sectionC, 100);

      /* ── Section D (Additional) ── */
      const pnpPts      = parseInt(document.getElementById('pnp').value) || 0;
      const jobPts      = parseInt(document.getElementById('jobOffer').value) || 0;
      const eduCanPts   = parseInt(document.getElementById('canadaEdu').value) || 0;
      const siblingPts  = parseInt(document.getElementById('sibling').value) || 0;
              const frenchPts   = parseInt(document.getElementById('frenchBonus').value) || 0;
      const sectionD = pnpPts + jobPts + eduCanPts + siblingPts + frenchPts;

      /* ── TOTAL ── */
      const totalScore = Math.min(1200, sectionA + sectionB + sectionC + sectionD);

      /* ── VERDICT ── */
      let verdict, verdictColor;
      if (pnpPts === 600) {
        verdict = '🏆 PNP Nominated — ITA Virtually Guaranteed!';
        verdictColor = '#1b5e20';
      } else if (totalScore >= 490) {
        verdict = '🟢 Excellent — Draw Ready';
        verdictColor = '#2e7d32';
      } else if (totalScore >= 460) {
        verdict = '🟡 Good — Competitive for CEC Draws';
        verdictColor = '#e65100';
      } else if (totalScore >= 430) {
        verdict = '🟠 Fair — May Qualify for CEC Draws';
        verdictColor = '#bf360c';
      } else {
        verdict = '🔴 Low — Improve Score First';
        verdictColor = '#b71c1c';
      }

      /* ── DISPLAY ── */
      document.getElementById('totalScore').textContent = totalScore;
      document.getElementById('verdict').textContent = verdict;
      const rh = document.getElementById('resultHeader');
      rh.style.background = verdictColor;
      document.querySelector('.result-box').style.borderColor = verdictColor;

      /* ── BREAKDOWN BARS ── */
      const rows = [
        ['Age', agePts, 110],
        ['Education', eduPts, 140],
        ['Language (CLB)', langPts, hasSpouse ? 128 : 136],
        ['Canadian Work Exp.', canExpPts, 80],
        ['Spouse Factors', sectionB, 40],
        ['Skill Transferability', sectionC, 100],
        ['Additional Points', sectionD, 700]
      ];
      document.getElementById('breakdownRows').innerHTML = rows.map(([label, pts, max]) => `
        <div class="br-row">
          <span class="br-label">${label}</span>
          <div class="br-bar-wrap">
            <div class="br-bar" style="width:${max > 0 ? Math.min(100, Math.round(pts / max * 100)) : 0}%"></div>
          </div>
          <span class="br-pts">${pts} / ${max}</span>
        </div>
      `).join('');

      /* ── TIPS ── */
      const tips = [];
      if (canExpPts === 0)  tips.push('🍁 You need at least 1 year of skilled Canadian work experience to qualify for CEC.');
      if (pnpPts === 0)     tips.push('🏔️ Apply to a PNP stream — a nomination adds +600 CRS, virtually guaranteeing an ITA.');
      if (avgClb < 9)       tips.push('📝 Improving to CLB 9 (IELTS 7.0+) adds 50–80 CRS points including transferability bonuses.');
      if (canExpPts < 64)   tips.push('⏱️ More Canadian work experience (3+ years) adds up to 64 direct CRS points.');
      if (frenchPts === 0)  tips.push('🇫🇷 Add French CLB 7+ with English CLB 5+ for +50 bonus points.');
      if (eduPts < 112)     tips.push('🎓 Get an ECA for your foreign degree — a bachelor\'s adds 112 education CRS points.');

      const tipsBox = document.getElementById('tipsBox');
      if (tips.length > 0) {
        tipsBox.style.display = 'block';
        tipsBox.innerHTML = `<h3>💡 Tips to Improve Your CEC CRS Score</h3><ul>${tips.map(t => `<li>${t}</li>`).join('')}</ul>`;
      } else {
        tipsBox.style.display = 'none';
      }

      const result = document.getElementById('result');
      result.style.display = 'block';
      result.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
    }

    /* RESET */
    function resetCalc() {
      document.getElementById('result').style.display = 'none';
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    /* SHARE */
    function shareResult() {
      const score = document.getElementById('totalScore').textContent;
      const url = window.location.href;
      if (navigator.share) {
        navigator.share({
          title: 'My CEC CRS Score 2026',
          text: `My CEC CRS score is ${score}/1200 — calculated free at CRS Score Calculator`,
          url
        });
      } else {
        navigator.clipboard.writeText(`My CEC CRS Score: ${score}/1200 — ${url}`)
          .then(() => alert('Result link copied to clipboard!'));
      }
    }

    /* CLOSE NAV ON OUTSIDE CLICK */
    document.addEventListener('click', function(e) {
      const nav = document.getElementById('navLinks');
      const toggle = document.querySelector('.nav-toggle');
      if (nav && nav?.classList.contains('open') && !nav.contains(e.target) && toggle && !toggle.contains(e.target)) {
        nav?.classList.remove('open');
        toggle.setAttribute('aria-expanded', 'false');
      }
    });
