// STEM NOC data
  const STEM_NOCS = {
    '20011':{ title:'Architecture and Science Managers', teer:0 },
    '21220':{ title:'Cybersecurity Specialists', teer:1 },
    '21300':{ title:'Civil Engineers', teer:1 },
    '21301':{ title:'Mechanical Engineers', teer:1 },
    '21310':{ title:'Electrical and Electronics Engineers', teer:1 },
    '21321':{ title:'Industrial and Manufacturing Engineers', teer:1 },
    '21331':{ title:'Geological Engineers', teer:1 },
    '22300':{ title:'Civil Engineering Technologists and Technicians', teer:2 },
    '22301':{ title:'Mechanical Engineering Technologists and Technicians', teer:2 },
    '22310':{ title:'Electrical and Electronics Engineering Technologists and Technicians', teer:2 },
    '63100':{ title:'Insurance Agents and Brokers', teer:3 }
  };
  const STEM_KEYWORDS = {
    'civil engineer':'21300','mechanical engineer':'21301','electrical engineer':'21310',
    'electronics engineer':'21310','industrial engineer':'21321','manufacturing engineer':'21321',
    'geological engineer':'21331','cybersecurity':'21220','cyber security':'21220',
    'architecture manager':'20011','science manager':'20011',
    'civil engineering technologist':'22300','civil technician':'22300',
    'mechanical technologist':'22301','mechanical technician':'22301',
    'electrical technologist':'22310','electrical technician':'22310',
    'insurance agent':'63100','insurance broker':'63100'
  };

  function checkEligibility() {
    const val = document.getElementById('nocInput').value.trim().toLowerCase();
    const res = document.getElementById('eligResult');
    if (!val) { res.style.display='none'; return; }
    let found = null;
    // Try exact NOC code
    if (STEM_NOCS[val]) { found = { noc: val, ...STEM_NOCS[val] }; }
    // Try keyword match
    if (!found) {
      for (const [kw, noc] of Object.entries(STEM_KEYWORDS)) {
        if (val.includes(kw)) { found = { noc, ...STEM_NOCS[noc] }; break; }
      }
    }
    if (found) {
      res.className = 'eligibility-result eligible-yes';
      res.innerHTML = `<strong>✅ STEM Eligible!</strong> NOC ${found.noc} — <strong>${found.title}</strong> (TEER ${found.teer}) qualifies for Express Entry STEM category draws. Calculate your CRS score below.`;
    } else {
      res.className = 'eligibility-result eligible-no';
      res.innerHTML = `<strong>❌ Not on STEM List.</strong> "${document.getElementById('nocInput').value}" is not among the 11 STEM-eligible NOC codes. You can still apply through <a href="/express-entry-crs-calculator">all-program Express Entry draws</a>. Check the NOC table above for eligible occupations.`;
    }
    res.style.display = 'block';
  }
  document.getElementById('nocInput').addEventListener('keypress', e => { if (e.key === 'Enter') checkEligibility(); });

  // Toggle helpers
  function toggleSpouse(el) {

  // CRS CALCULATION
  const AGE_PTS = {
    '17under':[0,0],'18':[99,90],'19':[105,95],'20':[110,100],'21':[110,100],'22':[110,100],
    '23':[110,100],'24':[110,100],'25':[110,100],'26':[110,100],'27':[110,100],'28':[110,100],
    '29':[110,100],'30':[105,95],'31':[99,90],'32':[94,85],'33':[88,80],'34':[83,75],
    '35':[77,70],'36':[72,65],'37':[66,60],'38':[61,55],'39':[55,50],'40':[50,45],
    '41':[39,35],'42':[28,25],'43':[17,15],'44':[6,5],'45plus':[0,0]
  };
  const EDU_PTS = {0:[0,0],1:[28,25],2:[84,75],3:[91,81],4:[112,100],5:[119,106],6:[126,112],7:[140,119]};
  const LANG1_PTS = {4:6,5:6,6:8,7:16,8:22,9:29,10:32};
  const LANG2_PTS = {4:0,5:1,6:1};
  const CANWORK_S = {0:0,1:40,2:53,3:64,4:72,5:80};
  const CANWORK_M = {0:0,1:35,2:46,3:56,4:63,5:70};
  const SP_EDU = {0:0,1:2,2:6,3:7,4:8,5:9,6:10,7:10};
  const SP_LANG = {4:0,5:1,6:1,7:3,8:3,9:5};
  const SP_WORK = {0:0,1:5,2:7,3:8,4:10,5:10};

  function getR(name) { const e = document.querySelector(`input[name="${name}"]:checked`); return e ? e.value : '0'; }

  function skillTransfer(fexp, langMin, edu, trades, eduLang) {
    let p = 0;
    if (langMin >= 7) { p += fexp === '1' ? 13 : fexp === '2' ? 25 : 0; }
    if (langMin >= 9) { p += fexp === '1' ? 12 : fexp === '2' ? 12 : 0; }
    if (eduLang === '1' && langMin >= 7) { p += parseInt(edu) >= 4 ? 25 : parseInt(edu) >= 2 ? 13 : 0; }
    if (fexp !== '0' && parseInt(edu) >= 4) { p += 13; if (fexp === '2') p += 12; }
    if (trades === '1' && langMin >= 5) { p += langMin >= 7 ? 50 : 25; }
    return Math.min(100, p);
  }

  function clbOf(id) { return parseInt(document.getElementById(id)?.value || '0') || 0; }

  function calculateCRS(ev) {
    if (ev && ev.preventDefault) ev.preventDefault();
    const married = getR('marital') === 'married';
    const spouseComing = getR('spouseComing') === 'yes';
    const withSpouse = married && spouseComing;
    const i = withSpouse ? 1 : 0;

    const ageSel = document.getElementById('age')?.value || '45plus';
    const age = (AGE_PTS[ageSel] || [0, 0])[i];

    const eduVal = document.getElementById('education')?.value || '0';
    const edu = (EDU_PTS[eduVal] || [0, 0])[i];

    const l1 = [clbOf('langSpeaking'), clbOf('langListening'), clbOf('langReading'), clbOf('langWriting')];
    const langMin = Math.min.apply(null, l1);
    const langMax = withSpouse ? 32 : 34;
    let lang1 = 0;
    l1.forEach(c => { lang1 += Math.min(LANG1_PTS[c] || 0, langMax); });

    const l2Used = (document.getElementById('lang2Test')?.value || 'none') !== 'none';
    let lang2 = 0;
    if (l2Used) {
      [clbOf('l2s'), clbOf('l2l'), clbOf('l2r'), clbOf('l2w')].forEach(c => {
        lang2 += c >= 9 ? (withSpouse ? 6 : 6) : c >= 7 ? 3 : c >= 5 ? 1 : 0;
      });
      lang2 = Math.min(lang2, withSpouse ? 22 : 24);
    }

    const canYears = parseInt(document.getElementById('canadianExp')?.value || '0') || 0;
    const canExp = (withSpouse ? CANWORK_M : CANWORK_S)[canYears] || 0;

    const coreA = age + edu + lang1 + lang2 + canExp;

    let coreB = 0;
    if (withSpouse) {
      coreB += SP_EDU[document.getElementById('spouseEdu')?.value || '0'] || 0;
      coreB += SP_WORK[parseInt(document.getElementById('spouseCanExp')?.value || '0') || 0] || 0;
      const spTest = (document.getElementById('spouseLangTest')?.value || 'none') !== 'none';
      if (spTest) {
        [clbOf('sls'), clbOf('sll'), clbOf('slr'), clbOf('slw')].forEach(c => {
          coreB += c >= 9 ? 5 : c >= 7 ? 3 : c >= 5 ? 1 : 0;
        });
      }
      coreB = Math.min(coreB, 40);
    }

    const fexp = document.getElementById('foreignExp')?.value || '0';
    const st = skillTransfer(fexp, langMin, eduVal, getR('tradesCert'), getR('eduLang'));

    let extra = 0;
    extra += parseInt(getR('pnp')) || 0;
    extra += parseInt(getR('jobOffer')) || 0;
    extra += parseInt(getR('jobOffer2')) || 0;
    extra += parseInt(getR('canEdu')) || 0;
    extra += parseInt(getR('canEdu2')) || 0;
    extra += parseInt(getR('sibling')) || 0;
    extra += parseInt(getR('frBonus')) || 0;
    extra = Math.min(extra, 600);

    const total = Math.min(1200, coreA + coreB + st + extra);

    const num = document.getElementById('scoreNumber');
    if (num) num.textContent = total;
    const bar = document.getElementById('scoreBar');
    if (bar) bar.style.width = Math.min(100, Math.round((total / 1200) * 100)) + '%';

    const verdict = document.getElementById('scoreVerdict');
    if (verdict) {
      verdict.textContent = total >= 520 ? 'Excellent — above recent all-program cutoffs.'
        : total >= 470 ? 'Strong — competitive in STEM category draws.'
        : total >= 400 ? 'Moderate — target a PNP nomination or improve language scores.'
        : 'Below typical cutoffs — see our improve-CRS guide.';
    }

    const bd = document.getElementById('scoreBreakdown');
    if (bd) {
      bd.innerHTML = [
        ['Core / human capital', coreA],
        ['Spouse factors', coreB],
        ['Skill transferability', st],
        ['Additional points', extra],
        ['Total CRS score', total]
      ].map(r => `<div class="bd-row"><span>${r[0]}</span><strong>${r[1]}</strong></div>`).join('');
    }

    const sv = document.getElementById('stemVerdictText');
    if (sv) {
      sv.textContent = total >= 470
        ? 'Your score is in range for recent STEM category-based draws (typically CRS 470–500).'
        : 'STEM draws have recently cut off near CRS 470–500. Boost language or gain Canadian experience.';
    }

    const results = document.getElementById('results');
    if (results) { results.style.display = 'block'; results.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); }
  }

  function copyScore() {
    const n = document.getElementById('scoreNumber');
    if (n && navigator.clipboard) navigator.clipboard.writeText('My STEM Express Entry CRS score: ' + n.textContent);
  }
  function shareWA() {
    const n = document.getElementById('scoreNumber');
    const t = encodeURIComponent('My STEM Express Entry CRS score is ' + (n ? n.textContent : '') + ' — check yours: https://crscalculator.site/stem-crs-calculator');
    window.open('https://wa.me/?text=' + t, '_blank');
  }
  function toggleFAQ(btn) {
    const ans = btn.nextElementSibling;
    if (ans) ans.classList.toggle('open');
    btn.classList.toggle('open');
    btn.setAttribute('aria-expanded', String(btn.classList.contains('open')));
  }
